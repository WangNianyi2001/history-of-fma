% Useful commands

% Split string by separator & wrap each with command
% Usage: \splitMap{separator}{command}{string}
\newcommand{\splitMap}[3]{{
	\setsepchar{#1}
	\readlist*\mylist{#3}
	\foreachitem\x\in\mylist[]{#2{\x}}
}}

% Importers

\newcommand{\importHistory}[1]{\subimport{histories}{#1.tex}}
\newcommand{\importStory}[1]{\subimport{stories}{#1.tex}}
\newcommand{\importPerson}[1]{\subimport{people}{#1.tex}}

% Inline styles

\newcommand{\name}[1]{{\hyperref[#1]{#1}}}
\newcommand{\person}[1]{\name{#1}}
\renewcommand{\thead}[1]{\textbf{ #1}}
\newcommand{\cname}[1]{{[\name{#1}]}}
\newcommand{\identity}[1]{{\small (#1)}}
